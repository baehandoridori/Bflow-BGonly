/** 그룹을 beforeGroupId 앞으로 이동 — 레일 드래그 순서 변경. beforeGroupId=null 이면 맨 뒤로. */
export function moveGroupBefore(groups: TabGroupLike[], groupId: string, beforeGroupId: string | null): TabGroupLike[] {
  if (groupId === beforeGroupId) return groups;
  const src = groups.find((g) => g.id === groupId);
  if (!src) return groups;
  const rest = groups.filter((g) => g.id !== groupId);
  if (beforeGroupId === null) return [...rest, src];
  const idx = rest.findIndex((g) => g.id === beforeGroupId);
  if (idx < 0) return groups;
  return [...rest.slice(0, idx), src, ...rest.slice(idx)];
}

### 3-3. 신규 `src/utils/dragAutoScroll.ts` (전체 — alias-free, 상대 import 없음)

```ts
/**
 * 드래그 자동 스크롤 (확정 시안 A) — 드래그 중 컨테이너 가장자리 근처에서 자동 스크롤.
 * computeEdgeScrollSpeed 는 순수 함수(단위 테스트 대상), bindVerticalDragAutoScroll 은 DOM 바인더.
 * 주의: node --test 가 직접 import 한다 — '@/' alias import 금지.
 */

/** 가장자리 근접 비례 속도. edge(px) 안에 들어오면 최대 max 까지 선형 증가. 음수=위로. */
export function computeEdgeScrollSpeed(pos: number, start: number, end: number, edge = 72, max = 14): number {
  const dStart = pos - start;
  const dEnd = end - pos;
  if (dStart < edge) return -max * (1 - Math.max(0, dStart) / edge);
  if (dEnd < edge) return max * (1 - Math.max(0, dEnd) / edge);
  return 0;
}

/** el 에 dragover 기반 세로 자동 스크롤을 바인딩. 해제 함수를 반환한다. rAF 루프는 드래그 중에만 돈다. */
export function bindVerticalDragAutoScroll(el: HTMLElement): () => void {
  let vy = 0;
  let raf = 0;
  let active = false;
  const tick = () => {
    if (vy !== 0) el.scrollTop += vy;
    raf = active ? requestAnimationFrame(tick) : 0;
  };
  const stop = () => {
    vy = 0;
    active = false;
    if (raf) { cancelAnimationFrame(raf); raf = 0; }
  };
  const onDragOver = (e: DragEvent) => {
    const r = el.getBoundingClientRect();
    vy = computeEdgeScrollSpeed(e.clientY, r.top, r.bottom);
    if (!active) { active = true; raf = requestAnimationFrame(tick); }
  };
  const onDragLeave = (e: DragEvent) => {
    // 자식 요소로 들어갈 때도 dragleave 가 발화한다 — 컨테이너를 실제로 벗어났을 때만 멈춘다.
    if (!el.contains(e.relatedTarget as Node | null)) stop();
  };
  el.addEventListener('dragover', onDragOver);
  el.addEventListener('drop', stop);
  el.addEventListener('dragleave', onDragLeave);
  document.addEventListener('dragend', stop);
  return () => {
    stop();
    el.removeEventListener('dragover', onDragOver);
    el.removeEventListener('drop', stop);
    el.removeEventListener('dragleave', onDragLeave);
    document.removeEventListener('dragend', stop);
  };
}
```

## 엣지 케이스

- **드롭 이중 발화:** 카드 onDrop → 섹션 onDrop 버블 — 기존 `draggingIdRef` 동기 소거가 그대로 방어(무수정). 레일은 섹션의 조상이 아니라 레일 드롭은 버블 무관.
- **접힌 레일:** DOM 동일 + CSS 만 좁아짐 — 드롭·클릭·하이라이트 전부 유지(목업 검증 동작). width 트랜지션 중 grid 는 auto-fill 리플로우로 자연 반응. 접힌 상태에선 ⠿ 이 숨어 순서 변경 불가(사양 — 펼쳐서 조작).
- **레일의 `transition-[width]` 는 의도적 예외:** 캐릭터 보드에는 "width 트랜지션 금지"(MO-11/F-2) 앵커 테스트가 있으나 대상 파일 고정 리스트(`characterBoardAssetWorkflow.test.ts:33-42`)에 레일 파일이 없어 충돌하지 않는다. 이는 **확정 시안 A 의 의도된 레이아웃 애니메이션**(접기/펼치기 한정, 200ms, reduced-motion 가드) — 리뷰에서 지적돼도 이 근거로 유지할 것.
- **scroll-spy 임계 130px:** sticky 헤더가 태그 펼침+칩 다줄로 130px 를 넘는 극단에서 하이라이트가 한 섹션 뒤처질 수 있음 — 동작 문제는 아니고 후속 조정 후보(부록 C).
- **sticky top 변동:** 태그 접기(작업 2)·칩 줄바꿈으로 헤더 높이 변화 → ResizeObserver 가 `--board-sticky-h` 갱신 → 레일 top·섹션 scroll-mt 동반 추적.
- **첫 로드/팝업:** 3-6(b) deps `[loaded]` 필수 — 미로드 조기 return 경로에서 `[]` 면 CSS 변수가 영영 미설정(레일이 헤더 밑에 깔림).
- **레일 카운트:** 현재 표시 대상(검색 적용) 기준 — 검색 시 그룹별 매칭 수가 보이고, 죽은 캐릭터 id 는 자연 제외.
- **미분류 위 그룹 드롭 = 맨 뒤로** — 미분류가 레일 마지막 항목이라 직관 일치.
- **per-card 인라인 드래그 콜백:** CQ-6 memo 최적화는 기존 구현의 알려진 트레이드오프 그대로(후속, 부록 C).
- **다른 창 동시 편집:** 그룹 순서 변경도 `updateTabGroups`(탭 단위 LWW) 경유 — 기존 수렴 규칙 그대로.
