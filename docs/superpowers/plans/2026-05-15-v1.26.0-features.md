```sql
CREATE POLICY "versions: anyone reads" ON scene_image_versions
  FOR SELECT USING (true);

CREATE POLICY "versions: insert authenticated" ON scene_image_versions
```

```ts
// src/utils/commentReactionUtils.ts
import type { CommentReaction, CommentReactionGroup } from '../types';

/**
 * 같은 이모지를 가진 리액션을 하나의 그룹으로 묶는다.
 * - 정렬: emoji 첫 등장 순서
 * - mine: 현재 사용자 user id 가 포함되어 있는지
 * - userNames: 누른 사람들의 이름 (툴팁용, 등록 순서)
 */
export function groupReactionsByEmoji(
  reactions: CommentReaction[],
  currentUserId: string | null,
): CommentReactionGroup[] {
  const map = new Map<string, CommentReactionGroup>();
  for (const r of reactions) {
    let g = map.get(r.emoji);
    if (!g) {
      g = { emoji: r.emoji, count: 0, userIds: [], userNames: [], mine: false };
      map.set(r.emoji, g);
    }
    if (g.userIds.includes(r.userId)) continue;   // 방어 (UNIQUE 제약으로 사실상 X)
    g.userIds.push(r.userId);
    g.userNames.push(r.userName);
    g.count++;
    if (currentUserId && r.userId === currentUserId) g.mine = true;
  }
  return Array.from(map.values());
}

/** 툴팁 텍스트 생성. 본인 있으면 "나" 강조, 최대 N명까지 표시. */
export function formatReactionTooltip(group: CommentReactionGroup, currentUserId: string | null): string {
  const maxShow = 5;
  const names = group.userIds.map((id, i) => (id === currentUserId ? '나' : group.userNames[i]));
  if (names.length <= maxShow) {
    return `${names.join(', ')} — 누른 사람: ${group.count}명`;
  }
  return `${names.slice(0, maxShow).join(', ')} 외 ${names.length - maxShow}명 — 누른 사람: ${group.count}명`;
}
```

```ts
// src/utils/imageVersionUtils.ts
import type { ImageVersion } from '../types';

export function pickLatestVersion(versions: ImageVersion[]): ImageVersion | null {
  if (versions.length === 0) return null;
  return versions.reduce((max, v) => v.versionNo > max.versionNo ? v : max);
}

export function nextVersionNo(versions: ImageVersion[]): number {
  if (versions.length === 0) return 1;
  return Math.max(...versions.map((v) => v.versionNo)) + 1;
}

/**
 * 삭제 가능 여부.
 * - 본인 + 마지막 1개 아닌 경우: OK
 * - 관리자 + 마지막 1개 아닌 경우: OK
 * - 그 외: X
 * onlyOne: 이 버전이 (씬 + 타입)의 유일한 버전인지
 */
export function canDeleteVersion(
  version: ImageVersion,
  currentUserId: string | null,
  isAdmin: boolean,
  onlyOne: boolean = false,
): boolean {
  if (onlyOne) return false;
  if (!currentUserId) return false;
  return version.createdBy === currentUserId || isAdmin;
}
```

```ts
    .from('comment_reactions')
    .insert({ comment_id: commentId, user_id: userId, emoji });

export async function removeCommentReaction(commentId: string, emoji: string, userId: string): Promise<void> {
  const { error } = await supabase
    .from('comment_reactions')

export async function getCommentReactionsBulk(commentIds: string[]): Promise<Record<string, { id: string; commentId: string; userId: string; userName: string; emoji: string; createdAt: string }[]>> {

  const { data, error } = await supabase
    .from('comment_reactions')
    .select('id, comment_id, user_id, emoji, created_at, users!inner(name)')
    .in('comment_id', commentIds);

  for (const row of data ?? []) {
    const cid = row.comment_id;

      commentId: row.comment_id,
      userId: row.user_id,
      userName: (row.users as any)?.name ?? '?',
      emoji: row.emoji,
      createdAt: row.created_at,
```

```ts
  const column = params.imageType === 'storyboard' ? 'storyboard_url' : 'guide_url';

  const after = remaining.filter((v) => v.id !== versionId);
  const latest = after.reduce((m, v) => v.versionNo > m.versionNo ? v : m);
```

```ts
      blob = await (await fetch(url)).blob();
    }
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = fileName;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(a.href);
    toast.success(`저장됨: ${fileName}`);
  } catch (err) {
    console.error('[downloadImage] 실패', err);
    toast.error('다운로드 실패');
  }
}

export async function copyImageToClipboard(url: string): Promise<void> {
  try {
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.src = url.startsWith('drive-img://') ? await driveImageToDataUrl(url) : url;
    await new Promise<void>((res, rej) => {
      img.onload = () => res();
      img.onerror = () => rej(new Error('이미지 로드 실패'));
    });
    const canvas = document.createElement('canvas');
    canvas.width = img.naturalWidth;
    canvas.height = img.naturalHeight;
    canvas.getContext('2d')!.drawImage(img, 0, 0);
    const pngBlob: Blob = await new Promise((res) => canvas.toBlob((b) => res(b!), 'image/png'));
    await navigator.clipboard.write([new ClipboardItem({ 'image/png': pngBlob })]);
    toast.success('클립보드에 이미지 복사됨');
  } catch (err) {
    console.error('[copyImageToClipboard] 실패', err);
    toast.error('이미지 복사 실패');
  }
}

export async function copyImageUrl(url: string): Promise<void> {
  try {
    await navigator.clipboard.writeText(url);
    toast.success('URL 복사됨');
  } catch {
    toast.error('URL 복사 실패');
  }
}

async function driveImageToDataUrl(driveUrl: string): Promise<string> {
  const buf: ArrayBuffer = await window.electronAPI.readDriveImage(driveUrl);
  const blob = new Blob([buf], { type: 'image/jpeg' });
  return URL.createObjectURL(blob);
}

/** 파일명 자동 생성: EP03_A_BG_007_storyboard_v2.jpg */
export function generateImageFileName(sheetName: string, sceneId: string, imageType: 'storyboard' | 'guide', versionNo: number): string {
  return `${sheetName}_${sceneId}_${imageType}_v${versionNo}.jpg`;
}
```

```tsx
import { useRef, useState, useEffect, useImperativeHandle, forwardRef } from 'react';
import { AnnotationToolbar, type DrawTool } from './AnnotationToolbar';

export interface AnnotationCanvasHandle {
  getCanvas(): HTMLCanvasElement | null;
  clear(): void;
}

interface Props {
  imageUrl: string;
  baseVersionNo: number;
}

export const AnnotationCanvas = forwardRef<AnnotationCanvasHandle, Props>(({ imageUrl, baseVersionNo }, ref) => {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const imgRef = useRef<HTMLImageElement>(null);
  const containerRef = useRef<HTMLDivElement>(null);
  const [tool, setTool] = useState<DrawTool>('pen');
  const [color, setColor] = useState('#EB5757');
  const [strokeWidth, setStrokeWidth] = useState(4);
  const [opacity, setOpacity] = useState(100);
  const [drawing, setDrawing] = useState(false);
  const startPos = useRef<{ x: number; y: number } | null>(null);
  const snapshot = useRef<ImageData | null>(null);
  const history = useRef<string[]>([]);
  const historyIndex = useRef(-1);

  // 캔버스 크기 = 이미지 자연 크기에 맞춤 (디스플레이는 CSS scale)
  useEffect(() => {
    const img = imgRef.current;
    if (!img || !canvasRef.current) return;
    const setSize = () => {
      const canvas = canvasRef.current!;
      canvas.width = img.naturalWidth;
      canvas.height = img.naturalHeight;
    };
    if (img.complete) setSize();
    else img.onload = setSize;
  }, [imageUrl]);

  useImperativeHandle(ref, () => ({
    getCanvas: () => canvasRef.current,
    clear: () => {
      const c = canvasRef.current; if (!c) return;
      c.getContext('2d')!.clearRect(0, 0, c.width, c.height);
      history.current = []; historyIndex.current = -1;
    },
  }));

  const getCtx = () => canvasRef.current?.getContext('2d') ?? null;

  const saveSnapshot = () => {
    const c = canvasRef.current; if (!c) return;
    history.current = history.current.slice(0, historyIndex.current + 1);
    history.current.push(c.toDataURL());
    historyIndex.current++;
    if (history.current.length > 50) history.current.shift();
  };

  const getCanvasPos = (e: React.MouseEvent) => {
    const c = canvasRef.current; if (!c) return { x: 0, y: 0 };
    const rect = c.getBoundingClientRect();
    return {
      x: (e.clientX - rect.left) * (c.width / rect.width),
      y: (e.clientY - rect.top) * (c.height / rect.height),
    };
  };

  const handleMouseDown = (e: React.MouseEvent) => {
    const ctx = getCtx(); if (!ctx) return;
    const p = getCanvasPos(e);
    ctx.strokeStyle = color;
    ctx.fillStyle = color;
    ctx.lineWidth = strokeWidth * (canvasRef.current!.width / (containerRef.current?.clientWidth ?? 1));
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';
    ctx.globalAlpha = tool === 'pen' ? (opacity / 100) * 0.5 : opacity / 100;
    ctx.globalCompositeOperation = tool === 'erase' ? 'destination-out' : 'source-over';

    if (tool === 'text') {
      const text = window.prompt('텍스트를 입력하세요:');
      if (text) {
        const fontSize = strokeWidth * 4 * (canvasRef.current!.width / (containerRef.current?.clientWidth ?? 1));
        ctx.font = `${fontSize}px Pretendard, sans-serif`;
        ctx.fillText(text, p.x, p.y);
        saveSnapshot();
      }
      return;
    }

    setDrawing(true);
    startPos.current = p;
    if (tool === 'pen' || tool === 'erase') {
      ctx.beginPath();
      ctx.moveTo(p.x, p.y);
    } else {
      snapshot.current = ctx.getImageData(0, 0, canvasRef.current!.width, canvasRef.current!.height);
    }
  };

  const handleMouseMove = (e: React.MouseEvent) => {
    if (!drawing) return;
    const ctx = getCtx(); if (!ctx || !startPos.current) return;
    const p = getCanvasPos(e);
    if (tool === 'pen' || tool === 'erase') {
      ctx.lineTo(p.x, p.y);
      ctx.stroke();
    } else if (snapshot.current) {
      ctx.putImageData(snapshot.current, 0, 0);
      drawShape(ctx, tool, startPos.current, p, ctx.lineWidth);
    }
  };

  const handleMouseUp = () => {
    if (!drawing) return;
    setDrawing(false);
    saveSnapshot();
  };

  const handleUndo = () => {
    if (historyIndex.current <= 0) {
      const c = canvasRef.current; if (c) c.getContext('2d')!.clearRect(0, 0, c.width, c.height);
      historyIndex.current = -1; return;
    }
    historyIndex.current--;
    const img = new Image();
    img.onload = () => {
      const c = canvasRef.current; if (!c) return;
      c.getContext('2d')!.clearRect(0, 0, c.width, c.height);
      c.getContext('2d')!.drawImage(img, 0, 0);
    };
    img.src = history.current[historyIndex.current];
  };

  // Ctrl+Z
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if ((e.ctrlKey || e.metaKey) && e.key === 'z' && !e.shiftKey) {
        e.preventDefault();
        handleUndo();
      }
    };
    document.addEventListener('keydown', handler);
    return () => document.removeEventListener('keydown', handler);
  }, []);

  return (
    <div ref={containerRef} className="relative w-full h-full flex items-center justify-center bg-[#0a0c12] overflow-hidden">
      <img ref={imgRef} src={imageUrl} className="max-w-full max-h-full select-none pointer-events-none" alt="" />
      <canvas
        ref={canvasRef}
        className="absolute top-0 left-0 w-full h-full"
        style={{ cursor: tool === 'text' ? 'text' : tool === 'erase' ? 'cell' : 'crosshair' }}
        onMouseDown={handleMouseDown}
        onMouseMove={handleMouseMove}
        onMouseUp={handleMouseUp}
        onMouseLeave={() => setDrawing(false)}
      />

      <AnnotationToolbar
        tool={tool}
        color={color}
        strokeWidth={strokeWidth}
        opacity={opacity}
        onToolChange={setTool}
        onColorChange={setColor}
        onStrokeChange={setStrokeWidth}
        onOpacityChange={setOpacity}
        onUndo={handleUndo}
        onClear={() => {
          const c = canvasRef.current; if (!c) return;
          c.getContext('2d')!.clearRect(0, 0, c.width, c.height);
          history.current = []; historyIndex.current = -1;
        }}
      />

      <div className="absolute bottom-3 left-3 text-[11px] text-text-muted bg-bg-primary/70 backdrop-blur px-2.5 py-1 rounded">
        <strong>대상</strong>: v{baseVersionNo} 위에 그리는 중 · <strong>저장 시</strong>: v{baseVersionNo + 1} (주석) 자동 생성
      </div>
    </div>
  );
});

function drawShape(ctx: CanvasRenderingContext2D, tool: DrawTool, start: {x:number;y:number}, end: {x:number;y:number}, baseWidth: number) {
  if (tool === 'rect') {
    ctx.strokeRect(start.x, start.y, end.x - start.x, end.y - start.y);
  } else if (tool === 'circle') {
    const rx = Math.abs(end.x - start.x) / 2;
    const ry = Math.abs(end.y - start.y) / 2;
    const cx = (start.x + end.x) / 2;
    const cy = (start.y + end.y) / 2;
    ctx.beginPath(); ctx.ellipse(cx, cy, rx, ry, 0, 0, Math.PI * 2); ctx.stroke();
  } else if (tool === 'line') {
    ctx.beginPath(); ctx.moveTo(start.x, start.y); ctx.lineTo(end.x, end.y); ctx.stroke();
  } else if (tool === 'arrow') {
    ctx.beginPath(); ctx.moveTo(start.x, start.y); ctx.lineTo(end.x, end.y); ctx.stroke();
    const angle = Math.atan2(end.y - start.y, end.x - start.x);
    const head = baseWidth * 3;
    ctx.beginPath();
    ctx.moveTo(end.x, end.y);
    ctx.lineTo(end.x - head * Math.cos(angle - Math.PI/6), end.y - head * Math.sin(angle - Math.PI/6));
    ctx.moveTo(end.x, end.y);
    ctx.lineTo(end.x - head * Math.cos(angle + Math.PI/6), end.y - head * Math.sin(angle + Math.PI/6));
    ctx.stroke();
  }
}
```

```ts
// src/utils/imageCompose.ts

/**
 * 원본 이미지 위에 캔버스를 합성해 JPEG blob 으로 반환.
 * 원본 이미지의 자연 크기를 보존한다 (캔버스도 같은 크기여야 함).
 */
export async function composeAnnotation(originalImageUrl: string, annotationCanvas: HTMLCanvasElement, quality: number = 0.9): Promise<Blob> {
  const img = new Image();
  img.crossOrigin = 'anonymous';
  img.src = originalImageUrl;
  await new Promise<void>((res, rej) => {
    img.onload = () => res();
    img.onerror = () => rej(new Error('이미지 로드 실패'));
  });

  const composite = document.createElement('canvas');
  composite.width = img.naturalWidth;
  composite.height = img.naturalHeight;
  const ctx = composite.getContext('2d')!;
  ctx.drawImage(img, 0, 0);
  ctx.drawImage(annotationCanvas, 0, 0);

  return new Promise((res, rej) => {
    composite.toBlob((b) => b ? res(b) : rej(new Error('toBlob 실패')), 'image/jpeg', quality);
  });
}
```
